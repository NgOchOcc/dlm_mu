\section{Proposed Method: Semantic-Ball Repulsion Unlearning (SBRU)}
\label{sec:method}

\subsection{Setup and notation}

\begin{itemize}
\item MDLM denoiser $p_\theta(\cdot \mid x, y_t)$ over vocabulary $\Vocab$; output (unembedding) matrix
$E \in \R^{|\Vocab| \times d}$ with rows $E_v$.
\item Forget pair $(x_f, y_f) \in \Df$; retain pair $(x_r, y_r) \in \Dr$; frozen reference $\theta_0$
(the pre-unlearning model).
\item Sample $t \sim \mathcal{U}[0,1]$, $y_t \sim q(\cdot \mid y_f, t)$, masked set $\Mt$.
\end{itemize}

\paragraph{Soft predicted embedding.} At masked position $i$,
\begin{equation}
\boxed{\;\ehat_\theta^{\,i} \;=\; E^\top p_\theta(\cdot \mid x_f, y_t)
\;=\; \sum_{v \in \Vocab} p_\theta(v \mid x_f, y_t)\, E_v \;\in\; \R^{d}. \;}
\label{eq:soft-emb}
\end{equation}
This is the probability-weighted centroid of token embeddings. It is the key object of the method: it
moves \emph{continuously} and encodes \emph{which region of meaning} the model is emitting, not merely
which exact token. It is differentiable in $\theta$ and costs one matrix product.

\paragraph{Forget-region centre.} $c^{i}$ is either the ground-truth forget-token embedding
$E_{y_f^{i}}$, or the reference soft embedding $\ehat_{\theta_0}^{\,i} = E^\top p_{\theta_0}(\cdot \mid
x_f, y_t)$. Optionally the region is a \textbf{semantic ball} spanned by the $k$ nearest neighbours of
$c^i$ under cosine similarity in $E$, which captures synonyms explicitly.

\paragraph{Kernel.} $\ksig(a,b) = \exp\!\big(-\|a - b\|^2 / 2\sigma^2\big) \in (0, 1]$.

\subsection{The loss}

SBRU is the discrete-diffusion, weight-time, embedding-space analogue of SGF's MMD repulsion. It has
three bounded terms.

\paragraph{(1) Bounded semantic repulsion (replaces GA).}
\begin{equation}
\boxed{\;
\Lrep(\theta) \;=\; \E_{(x_f,y_f),\,t,\,y_t}
\Big[\; \frac{1}{|\Mt|} \sum_{i \in \Mt} \ksig\big(\ehat_\theta^{\,i},\, c^{i}\big) \Big]
\;\in\; [0,1]. \;}
\label{eq:lrep}
\end{equation}
Minimizing drives the soft embedding \emph{out of the $\sigma$-ball} around the forget region. Its
gradient is exactly SGF's repulsive field,
\begin{equation}
\nabla_{\ehat}\, \ksig(\ehat, c) \;=\; -\frac{1}{\sigma^2}\, \ksig(\ehat, c)\,(\ehat - c),
\label{eq:lrep-grad}
\end{equation}
which \textbf{vanishes once} $\|\ehat - c\| \gg \sigma$ --- a built-in stopping point. Because
$\ksig \in [0,1]$, the objective is bounded below by $0$: there is no $-\infty$ to chase, hence no
collapse.

\paragraph{(2) Generic-anchor attraction (gives the model somewhere safe to go).}
\begin{equation}
\Lanc(\theta) \;=\; -\,\E\Big[\frac{1}{|\Mt|}\sum_{i\in\Mt} \ksig\big(\ehat_\theta^{\,i},\, a^{i}\big)\Big]
\;\in\; [-1, 0],
\qquad
a^{i} = E^\top p_{\theta_0}(\cdot \mid m,\, y_t),
\label{eq:lanc}
\end{equation}
where $a^i$ is the \textbf{prompt-masked unconditional soft embedding} (borrowing MDU's anchor) --- the
model's generic linguistic prior region. This prevents the degenerate gibberish GA produces, and is
itself bounded.

\paragraph{(3) Retain regularizer.}
\begin{equation}
\Lret(\theta) \;=\; \Lsft(\theta; \Dr)
\quad\text{(optionally }+\;\E_{\Dr}\big[\|\ehat_\theta^{\,i} - \ehat_{\theta_0}^{\,i}\|^2\big]\text{, RMU/KUDA-style).}
\label{eq:lret}
\end{equation}

\paragraph{Total objective.}
\begin{equation}
\boxed{\;\mathcal{L}_{\mathrm{SBRU}} \;=\; \Lrep \;+\; \gamma\,\Lanc \;+\; \lambda\,\Lret. \;}
\label{eq:total}
\end{equation}

\subsection{Why this satisfies the three requirements}

\begin{keybox}
\textbf{R2 --- why synonym leakage is fixed.} Emitting a synonym $v'$ with $E_{v'} \approx c$ keeps
$\ehat_\theta$ \emph{inside} the ball, so $\ksig(\ehat, c)$ stays near $1$ and the loss is \emph{not}
reduced. The only way to lower $\Lrep$ is to move the entire soft embedding to a semantically distant
region --- which forbids all near-neighbours of the forget token \emph{simultaneously}. GA, by contrast
(D1), lowers only $p(y_f)$ and lets the freed mass flow to exactly those neighbours. Using the $k$-NN
semantic ball for $c^i$ makes this explicit.

\medskip
\textbf{R3 --- why it does not collapse.} (i) $\Lrep$ is bounded in $[0,1]$ with vanishing gradient
outside the ball, per~\eqref{eq:lrep-grad}; (ii) $\Lanc$ supplies a finite attraction target; (iii)
$\Lret$ anchors utility. This mirrors MDU's insight that a \emph{target} prevents divergence, but
implements it in embedding space so the target is semantic rather than token-level.

\medskip
\textbf{R1 --- simplicity.} Two extra lines over the standard SFT loop: one matrix product for
$\ehat$, one exponential. No architectural change, no null-space projection, no multi-stage pipeline.
\end{keybox}
