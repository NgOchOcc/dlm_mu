\begin{abstract}
This note compares the SimpleTSD multi-draw training loop \emph{before} and
\emph{after} the out-of-memory (OOM) fix. Multi-draw training averages the
per-example loss over $M=\texttt{loss\_mc}$ independent partial-mask draws to
lower the pathwise gradient variance. The buggy version accumulated the loss of
all $M$ draws into one scalar and backpropagated \emph{once}, so all $M$ student
computation graphs stayed alive simultaneously and activation memory scaled as
$O(M)$, triggering OOM under DeepSpeed ZeRO-3. The fixed version backpropagates
\emph{each draw immediately} and accumulates gradients in DeepSpeed's buffer, so
only one graph is alive at a time and activation memory is $O(1)$ in $M$. We show
the two versions realise the \emph{same} optimizer update
(Proposition~\ref{prop:equiv}); only the peak memory and the number of backward
calls differ.
\end{abstract}

\section{Shared objective}

Both versions optimise the identical objective. Fix one minibatch example
$(x,y,\Ahat)$ with a nonempty selected forget set $\Ahat$. For a single mask draw
$\Mset$ with masked forget positions $F=\Mset\cap\Ahat$ and self-distillation
positions $R=\Mset\setminus\Ahat$, the per-draw per-example loss is
\begin{equation}
  \Lex(\Mset)=
  \begin{cases}
    \alpha\,\mathcal L_{\mathrm f}(\Mset)+(1-\alpha)\,\mathcal L_{\mathrm d}(\Mset), & R\neq\varnothing,\\[2pt]
    \mathcal L_{\mathrm f}(\Mset), & R=\varnothing,
  \end{cases}
  \label{eq:perdraw}
\end{equation}
where $\mathcal L_{\mathrm f}(\Mset)=\tfrac{1}{|F|}\sum_{i\in F}\KL(q_{\Mset,i}\|p^\theta_{\Mset,i})$
is the demotion term (gold token removed from the teacher and renormalised) and
$\mathcal L_{\mathrm d}(\Mset)=\tfrac{1}{|R|}\sum_{i\in R}\KL(p^0_{\Mset,i}\|p^\theta_{\Mset,i})$
is the self-distillation term. The multi-draw example loss averages $M$
independent draws $\Mset^{(1)},\ldots,\Mset^{(M)}$:
\begin{equation}
  \mathcal L(x,y,\Ahat)\;=\;\frac{1}{M}\sum_{m=1}^{M}\Lex\!\big(\Mset^{(m)}\big).
  \label{eq:mcloss}
\end{equation}
The selected set $\Ahat$ (PCSS, run once per example) and, in \texttt{cover}
mode, the joint mask plan $\{\Mset^{(m)}\}$ are \emph{mask-independent context}:
computed once and shared by every draw. Only the student and teacher forward
passes are repeated per draw.

\section{Version A (before): accumulate then backpropagate once}

\begin{algorithm}[h]
\caption{\bad{Buggy} multi-draw step --- one backward over $M$ live graphs}
\label{alg:before}
\small
\begin{algorithmic}[1]
\Require Example context (inputs, $\Ahat$, cover plan); draws $M$.
\State $\ell \gets 0$ \Comment{scalar accumulator holding \bad{all $M$ graphs}}
\For{$m=1,\ldots,M$}
  \State Draw $\Mset^{(m)}$; build $y_{\Mset^{(m)}}$.
  \State \bad{$o^{(m)}\gets \text{student}(y_{\Mset^{(m)}})$} \Comment{graph $m$ kept alive}
  \State $p^0\gets\text{teacher}(y_{\Mset^{(m)}})$ \Comment{under \texttt{no\_grad}}
  \State $\ell \gets \ell + \Lex(\Mset^{(m)})$ \Comment{\bad{graphs $1..m$ still alive}}
\EndFor
\State $\ell \gets \ell / M$
\State $\texttt{accelerator.backward}(\gamma\,\ell)$ \Comment{\bad{one backward, $M$ graphs}}
\end{algorithmic}
\end{algorithm}

Because $\ell$ references the output of every student forward, autograd cannot
free any draw's activations until the single \texttt{backward} at the end. Peak
activation memory is therefore
\begin{equation}
  \text{Mem}_{\text{A}} \;=\; O(M)\times(\text{one student forward}),
\end{equation}
which OOMs under ZeRO-3 for $M>1$ once the sharded activations, offload buffers,
and the teacher forward are added.

\section{Version B (after): per-draw backward, gradient accumulation}

\begin{algorithm}[h]
\caption{\good{Fixed} multi-draw step --- one backward \emph{per draw}}
\label{alg:after}
\small
\begin{algorithmic}[1]
\Require Example context; draws $M$; HF boundary flag $\textsc{sync}=$\,\texttt{accelerator.sync\_gradients}.
\State $s_0 \gets \textsc{sync}$ \Comment{save HF's boundary for this micro-step}
\For{$m=1,\ldots,M$}
  \State \good{set\_sync\_gradients}$\big(s_0 \wedge (m=M)\big)$ \Comment{boundary only on the last draw}
  \State Draw $\Mset^{(m)}$; build $y_{\Mset^{(m)}}$.
  \State $o^{(m)}\gets \text{student}(y_{\Mset^{(m)}})$
  \State $p^0\gets\text{teacher}(y_{\Mset^{(m)}})$ \Comment{under \texttt{no\_grad}}
  \State $\ell_m \gets \gamma\,\Lex(\Mset^{(m)})\,/\,M$
  \State \good{$\texttt{accelerator.backward}(\ell_m)$} \Comment{frees graph $m$; grads sum}
\EndFor
\State \good{set\_sync\_gradients}$(s_0)$ \Comment{restore HF's view for the outer loop}
\end{algorithmic}
\end{algorithm}

Each \texttt{backward} frees draw $m$'s graph before draw $m{+}1$ runs, so
\begin{equation}
  \text{Mem}_{\text{B}} \;=\; O(1)\times(\text{one student forward}),
  \qquad\text{independent of } M.
\end{equation}

\paragraph{Why the boundary flag matters (DeepSpeed ZeRO-3).}
Under \texttt{accelerate 1.11.0}, \texttt{accelerator.backward} forwards
\texttt{sync\_gradients} into the DeepSpeed engine wrapper's \texttt{backward},
which
\begin{enumerate}\itemsep2pt
  \item sets DeepSpeed's gradient-accumulation boundary explicitly
        (via \texttt{set\_gradient\_}\allowbreak\texttt{accumulation\_}\allowbreak\texttt{boundary}),
        overriding its internal micro-step counter; and
  \item calls \texttt{engine.step()} (reduce-scatter, clip, optimizer step, zero-grad,
        LR step) \emph{only} when \texttt{sync\_gradients} is true.
\end{enumerate}
Forcing the flag false on draws $1..M{-}1$ accumulates their gradients into
DeepSpeed's buffer with \emph{no} reduce/step; the last draw performs the reduce
and the single optimizer step --- but only on HF's own accumulation boundary
($s_0$ true). On non-boundary micro-steps every draw stays false and merely
accumulates, exactly as ordinary gradient accumulation across micro-batches. The
outer HF \texttt{optimizer.step()} is a no-op under DeepSpeed, so there is no
double step.

\section{Equivalence of the two versions}

\begin{proposition}[Identical optimizer update]
\label{prop:equiv}
For fixed draws $\{\Mset^{(m)}\}$, Versions~A and~B produce the same accumulated
parameter gradient, hence the same optimizer step.
\end{proposition}

\begin{proof}
The teacher forward is under \texttt{no\_grad}, so $q_{\Mset,i}$ and
$p^0_{\Mset,i}$ are constants w.r.t.\ $\theta$; the only $\theta$-dependence is
through $p^\theta_{\Mset^{(m)}}$, and distinct draws share no activation.
Version~A backpropagates $\gamma\,\tfrac{1}{M}\sum_m\Lex(\Mset^{(m)})$ once,
Version~B backpropagates $\ell_m=\gamma\,\Lex(\Mset^{(m)})/M$ separately.
PyTorch/DeepSpeed \emph{accumulate} (sum) gradients across \texttt{backward}
calls with no intervening reduce or zero-grad (boundary false on $m<M$), so
\begin{equation*}
  g_B \;=\; \sum_{m=1}^{M}\nabla_\theta\ell_m
     \;=\; \frac{\gamma}{M}\sum_{m=1}^{M}\nabla_\theta\Lex(\Mset^{(m)})
     \;=\; g_A .
\end{equation*}
Because $g_A=g_B$ and the reduce-scatter is linear, both trigger one identical
\texttt{engine.step()} on the boundary. The explicit $1/M$ scaling is orthogonal
to DeepSpeed's own $1/\texttt{gradient\_accumulation\_steps}$ scaling (accelerate
does not divide the loss for DeepSpeed), so relative scale versus the single-draw
path is preserved.
\end{proof}

\begin{remark}[Not bit-identical]
Versions~A and~B are mathematically equal but need not be bit-identical:
summation order in gradient accumulation and the timing of RNG draws differ.
For $M=1$ the fixed code takes the original single-draw path
(\texttt{super().training\_step}) verbatim, so that case is unchanged.
\end{remark}
