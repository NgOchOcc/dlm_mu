\section{Training Objective}

AURA-DLM removes the association between the forget prompt and responses
that the current model can still recover, while preserving the original
model behavior under a minimal prompt intervention.

For a response $z$ and corrupted state $z_t$, define the association excess
at masked position $i$ as
\begin{equation}
a_{\theta,i}(x,z_t)
=
\log p_{\theta}(z_i \mid x,z_t)
-
\log p_{\theta_0}(z_i \mid x^{-},z_t).
\end{equation}

The erase loss penalizes only association above margin $\delta$:
\begin{equation}
\mathcal{L}_{\mathrm{erase}}
=
\mathbb{E}
\left[
\frac{1}{|\mathcal{M}_t|}
\sum_{i \in \mathcal{M}_t}
\left[a_{\theta,i}(x,z_t)-\delta\right]_+
\right],
\label{eq:erase}
\end{equation}
where $[u]_+=\max(u,0)$.

The preservation loss distills the frozen model on the intervened prompt:
\begin{equation}
\mathcal{L}_{\mathrm{preserve}}
=
\mathbb{E}
\left[
\frac{1}{|\mathcal{M}_t|}
\sum_{i \in \mathcal{M}_t}
D_{\mathrm{KL}}
\left(
p_{\theta_0}(\cdot \mid x^{-},z_t)
\;\Vert\;
p_{\theta}(\cdot \mid x^{-},z_t)
\right)
\right].
\label{eq:preserve}
\end{equation}

The final objective is
\begin{equation}
\boxed{
\mathcal{L}_{\mathrm{AURA}}
=
\mathcal{L}_{\mathrm{erase}}
+
\lambda\mathcal{L}_{\mathrm{preserve}}.
}
\label{eq:total}
\end{equation}

\section{Finding the Forget-Key Intervention}

For each forget pair $(x,y)$, find a small prompt span whose removal most
reduces reconstruction of the forget response.

For candidate span $s$, let $x_{\setminus s}$ denote the prompt after masking
or deleting $s$. Define
\begin{equation}
A(s)
=
\mathbb{E}_{t,y_t}
\left[
\frac{1}{|\mathcal{M}_t|}
\sum_{i\in\mathcal{M}_t}
\left(
\log p_{\theta_0}(y_i\mid x,y_t)
-
\log p_{\theta_0}(y_i\mid x_{\setminus s},y_t)
\right)
\right].
\end{equation}

Select
\begin{equation}
s^\star
=
\arg\max_{s\in\mathcal{S}(x)}
\frac{A(s)}{|s|},
\qquad
x^{-}=x_{\setminus s^\star},
\label{eq:keyspan}
\end{equation}
where $\mathcal{S}(x)$ is a small set of candidate token spans.

\section{Algorithms}

\begin{algorithm}[t]
\caption{Precompute minimal prompt interventions}
\label{alg:intervention}
\begin{algorithmic}[1]
\Require Frozen MDLM $\theta_0$, forget dataset $\mathcal{D}_f$, maximum span length $L_{\max}$, number of mask samples $R$
\Ensure One intervened prompt $x^{-}$ for every $(x,y)\in\mathcal{D}_f$

\ForAll{$(x,y)\in\mathcal{D}_f$}
    \State Build candidate spans $\mathcal{S}(x)=\{s:1\le |s|\le L_{\max}\}$
    \State $bestScore \gets -\infty$, $x^{-}\gets x$

    \ForAll{$s\in\mathcal{S}(x)$}
        \State $score\gets 0$
        \For{$r=1$ to $R$}
            \State Sample diffusion level $t$
            \State Sample masked response $y_t$ and masked positions $\mathcal{M}_t$
            \ForAll{$i\in\mathcal{M}_t$}
                \State $score\gets score + \log p_{\theta_0}(y_i\mid x,y_t)-\log p_{\theta_0}(y_i\mid x_{\setminus s},y_t)$
            \EndFor
        \EndFor
        \State $score\gets score/(R\cdot |s|)$
        \If{$score>bestScore$}
            \State $bestScore\gets score$
            \State $x^{-}\gets x_{\setminus s}$
        \EndIf
    \EndFor
    \State Cache $(x,y,x^{-})$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{AURA-DLM training}
\label{alg:training}
\begin{algorithmic}[1]
\Require Frozen teacher $\theta_0$, trainable model $\theta\leftarrow\theta_0$, cached triples $(x,y,x^{-})$, margin $\delta$, weight $\lambda$, on-policy sample count $K$
\Ensure Unlearned MDLM parameters $\theta$

\While{not converged}
    \State Sample forget triple $(x,y,x^{-})$
    \State Initialize response set $\mathcal{B}\gets\{y\}$

    \For{$k=1$ to $K$}
        \State Sample complete response $z^{(k)}\sim p_{\theta}(\cdot\mid x)$
        \State $\mathcal{B}\gets\mathcal{B}\cup\{z^{(k)}\}$
    \EndFor

    \State $\mathcal{L}\gets 0$

    \ForAll{$z\in\mathcal{B}$}
        \State Sample diffusion level $t$
        \State Corrupt $z$ to obtain $z_t$ and masked positions $\mathcal{M}_t$

        \State Teacher forward: $q_i(\cdot)\gets p_{\theta_0}(\cdot\mid x^{-},z_t)$ for all $i\in\mathcal{M}_t$
        \State Student full-prompt forward: $p_i^{+}(\cdot)\gets p_{\theta}(\cdot\mid x,z_t)$
        \State Student intervention forward: $p_i^{-}(\cdot)\gets p_{\theta}(\cdot\mid x^{-},z_t)$

        \State $\mathcal{L}_{erase}\gets0$, $\mathcal{L}_{preserve}\gets0$

        \ForAll{$i\in\mathcal{M}_t$}
            \State $a_i\gets \log p_i^{+}(z_i)-\log q_i(z_i)$
            \State $\mathcal{L}_{erase}\gets\mathcal{L}_{erase}+\max(a_i-\delta,0)$
            \State $\mathcal{L}_{preserve}\gets\mathcal{L}_{preserve}+D_{\mathrm{KL}}(q_i\Vert p_i^{-})$
        \EndFor

        \State $\mathcal{L}_{erase}\gets\mathcal{L}_{erase}/|\mathcal{M}_t|$
        \State $\mathcal{L}_{preserve}\gets\mathcal{L}_{preserve}/|\mathcal{M}_t|$
        \State $\mathcal{L}\gets\mathcal{L}+\mathcal{L}_{erase}+\lambda\mathcal{L}_{preserve}$
    \EndFor

    \State $\mathcal{L}\gets\mathcal{L}/|\mathcal{B}|$
    \State Update $\theta$ with one optimizer step on $\nabla_\theta\mathcal{L}$
\EndWhile
\end{algorithmic}
\end{algorithm}

\section{Implementation Notes}

\paragraph{Masking.}
Use the same forward corruption process as the target MDLM backbone, e.g.,
LLaDA or Dream. Evaluate the loss only at masked response positions.

\paragraph{On-policy sampling.}
Always include ground-truth response $y$. Sample $K$ additional responses
from the current model and refresh them during training so newly emerging
paraphrases can become training targets.

\paragraph{Intervention.}
Masking the selected span is preferable when the backbone already uses a mask
token. Use the same intervention rule in precomputation and training.

\paragraph{Teacher gradients.}
All outputs from $\theta_0$ are stop-gradient.

\paragraph{Stable implementation.}
Use \texttt{log\_softmax} for token log probabilities. For the preservation
term, compute teacher-to-student KL exactly as
$D_{\mathrm{KL}}(p_{\theta_0}\Vert p_\theta)$.

\paragraph{Batching.}
Concatenate student conditions $(x,z_t)$ and $(x^{-},z_t)$ along the batch
dimension. The frozen teacher only requires $(x^{-},z_t)$.

\paragraph{Minimal initial setting.}
A simple first implementation can use
\[
K=1,\qquad L_{\max}=4,\qquad R=4,
\]
and tune only $\lambda$ and $\delta$. Avoid semantic classifiers, auxiliary
LLMs, representation losses, or adversarial RL until the base objective has
been tested.
