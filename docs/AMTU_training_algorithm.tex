\section{Training Objective}

For a forget example $(x,y)$, let $s_M^y$ denote the response $y$ with the
positions in $M$ replaced by $\Mask$.  At a masked position $i\in M$, define
\begin{equation}
    \ell_{\theta,i}(x,s_M^y)
    =
    \KL\!\left(
        p_{\theta,i}(\cdot\mid x,s_M^y)
        \,\middle\|\,
        a_i(\cdot\mid x,s_M^y)
    \right),
    \label{eq:local-kl}
\end{equation}
where $a_i$ is a frozen counterfactual or forget-free anchor distribution.
The AMTU objective is
\begin{equation}
\min_{\theta}\;
\mathbb{E}_{(x,y)\sim\Df}
\left[
    \max_{M\in\Mset}
    \max_{i\in M\cap\Iforget(x,y)}
    \ell_{\theta,i}(x,s_M^y)
\right]
+
\lambda\,\mathcal{R}_{\mathrm{retain}}(\theta),
\label{eq:amtu-objective}
\end{equation}
with retain trust region
\begin{equation}
\mathcal{R}_{\mathrm{retain}}(\theta)
=
\mathbb{E}_{\substack{(x_r,y_r)\sim\Dr\\M_r\sim Q_r}}
\left[
    \frac{1}{|M_r|}
    \sum_{i\in M_r}
    \KL\!\left(
        p_{\theta_0,i}(\cdot\mid x_r,s_{M_r}^{y_r})
        \,\middle\|\,
        p_{\theta,i}(\cdot\mid x_r,s_{M_r}^{y_r})
    \right)
\right].
\label{eq:retain-kl}
\end{equation}

\section{Discrete Adversarial Mask Search}

For a fixed mask cardinality $k$, the inner maximization is approximated by
cardinality-preserving mask swaps.  Define the smooth surrogate
\begin{equation}
    h_{\beta}(M)
    =
    \frac{1}{\beta}
    \log
    \sum_{i\in M\cap\Iforget(x,y)}
    \exp\!\left(\beta\,\ell_{\theta,i}(x,s_M^y)\right),
    \label{eq:smooth-max}
\end{equation}
which approaches $\max_i\ell_{\theta,i}$ as $\beta\to\infty$.
Let $E_j$ be the input embedding at response position $j$ and
$g_j=\nabla_{E_j}h_\beta(M)$.  Swapping a masked position $u\in M$ with a
visible position $v\notin M$ has first-order score
\begin{equation}
\Delta_{u,v}
\approx
    g_u^{\top}\!\left(e(y_u)-e_{\mathrm{mask}}\right)
    +
    g_v^{\top}\!\left(e_{\mathrm{mask}}-e(y_v)\right).
\label{eq:swap-score}
\end{equation}
The proposed swap is accepted only when it increases the exact discrete loss.

\begin{algorithm}[t]
\caption{Cardinality-Preserving Adversarial Mask Search}
\label{alg:mask-search}
\begin{algorithmic}[1]
\Require Current model $p_\theta$; frozen anchor $a$; forget pair $(x,y)$;
         sensitive positions $\Iforget(x,y)$; mask size $k$; restarts $R$;
         swap steps $q$; smoothing parameter $\beta$
\Ensure Adversarial mask $M^{\star}$, attacked position $i^{\star}$, and loss $L^{\star}$
\State $\mathcal{C}\gets\varnothing$
\For{$r=1,\ldots,R$}
    \State Sample $M\subseteq[|y|]$ uniformly subject to $|M|=k$
    \For{$t=1,\ldots,q$}
        \State Construct the valid MDLM state $s_M^y$
        \State Compute $\ell_{\theta,i}(x,s_M^y)$ for all $i\in M\cap\Iforget(x,y)$ using Eq.~\eqref{eq:local-kl}
        \State Compute $h_\beta(M)$ from Eq.~\eqref{eq:smooth-max}
        \State Backpropagate $h_\beta(M)$ to obtain $g_j=\nabla_{E_j}h_\beta(M)$ for all response positions $j$
        \State $(u^{\star},v^{\star})\gets
        \displaystyle\arg\max_{u\in M,\,v\notin M}\Delta_{u,v}$ using Eq.~\eqref{eq:swap-score}
        \State $\widetilde{M}\gets(M\setminus\{u^{\star}\})\cup\{v^{\star}\}$
        \State $L(M)\gets\displaystyle\max_{i\in M\cap\Iforget(x,y)}\ell_{\theta,i}(x,s_M^y)$
        \State $L(\widetilde{M})\gets\displaystyle\max_{i\in \widetilde{M}\cap\Iforget(x,y)}\ell_{\theta,i}(x,s_{\widetilde{M}}^y)$
        \If{$L(\widetilde{M})>L(M)$}
            \State $M\gets\widetilde{M}$
        \Else
            \State \textbf{break}
        \EndIf
    \EndFor
    \State $i\gets\displaystyle\arg\max_{j\in M\cap\Iforget(x,y)}\ell_{\theta,j}(x,s_M^y)$
    \State Add $(M,i,\ell_{\theta,i}(x,s_M^y))$ to $\mathcal{C}$
\EndFor
\State $(M^{\star},i^{\star},L^{\star})\gets\displaystyle\arg\max_{(M,i,L)\in\mathcal{C}}L$
\State \Return $(M^{\star},i^{\star},L^{\star})$
\end{algorithmic}
\end{algorithm}

\section{AMTU Training}

The outer loop searches several mask-cardinality strata so that early, middle,
and late denoising states are all covered.  The theoretical default commits one
token at a time.  A practical top-$b$ variant can replace the single maximum by
$\TopBMean$, but the one-token form gives the cleanest transition-level guarantee.

\begin{algorithm}[t]
\caption{Adversarial Masked-Transition Unlearning (AMTU)}
\label{alg:amtu}
\begin{algorithmic}[1]
\Require Initial MDLM parameters $\theta_0$; frozen anchor $a$; forget data $\Df$;
         retain data $\Dr$; mask-cardinality strata $\mathcal{K}$; restarts $R$;
         swap steps $q$; retain weight $\lambda$; learning rate $\eta$;
         number of updates $T$
\Ensure Unlearned parameters $\theta$
\State $\theta\gets\theta_0$
\For{$t=1,\ldots,T$}
    \State Sample forget minibatch $B_f\subset\Df$ and retain minibatch $B_r\subset\Dr$
    \State $\mathcal{L}_f\gets0$
    \ForAll{$(x,y)\in B_f$}
        \State $\mathcal{A}\gets\varnothing$
        \ForAll{$k\in\mathcal{K}$}
            \State $(M_k^{\star},i_k^{\star},L_k^{\star})
            \gets \Call{AdversarialMaskSearch}{\theta,a,x,y,\Iforget(x,y),k,R,q}$
            \State Add $(M_k^{\star},i_k^{\star},L_k^{\star})$ to $\mathcal{A}$
        \EndFor
        \State $(M^{\star},i^{\star},L^{\star})
        \gets\displaystyle\arg\max_{(M,i,L)\in\mathcal{A}}L$
        \State $\mathcal{L}_f\gets\mathcal{L}_f+
        \KL\!\left(
            p_{\theta,i^{\star}}(\cdot\mid x,s_{M^{\star}}^y)
            \,\middle\|\,
            a_{i^{\star}}(\cdot\mid x,s_{M^{\star}}^y)
        \right)$
    \EndFor
    \State $\mathcal{L}_f\gets\mathcal{L}_f/|B_f|$
    \State $\mathcal{L}_r\gets0$
    \ForAll{$(x_r,y_r)\in B_r$}
        \State Sample retain mask $M_r\sim Q_r$
        \State $\mathcal{L}_r\gets\mathcal{L}_r+
        \displaystyle\frac{1}{|M_r|}\sum_{i\in M_r}
        \KL\!\left(
            p_{\theta_0,i}(\cdot\mid x_r,s_{M_r}^{y_r})
            \,\middle\|\,
            p_{\theta,i}(\cdot\mid x_r,s_{M_r}^{y_r})
        \right)$
    \EndFor
    \State $\mathcal{L}_r\gets\mathcal{L}_r/|B_r|$
    \State $\theta\gets\theta-\eta\nabla_\theta\left(\mathcal{L}_f+\lambda\mathcal{L}_r\right)$
    \Comment{Stop gradients through $a$, $M^{\star}$, and $i^{\star}$}
\EndFor
\State \Return $\theta$
\end{algorithmic}
\end{algorithm}

\paragraph{Implementation note.}
The frozen anchor logits may be cached when the anchor and the candidate state
are fixed.  In large-scale training, the $O(n^2)$ exhaustive swap scoring in
Algorithm~\ref{alg:mask-search} can be approximated by selecting a small set of
high-scoring masked and visible positions before evaluating pairwise swaps.
Every accepted candidate is nevertheless evaluated as a genuine discrete MDLM
state rather than a continuous soft-mask relaxation.
